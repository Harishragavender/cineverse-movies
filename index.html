🗄 DATABASE SCHEMA (SQLite • parameterized queries • FK cascades)

users
id INTEGER PK
name TEXT
email TEXT UNIQUE
password TEXT (hashed)
movies
id INTEGER PK
title, genre, year
rating REAL, duration
description, poster_url
trailer_url
favorites
id INTEGER PK
user_id → users.id
movie_id → movies.id
UNIQUE(user_id, movie_id)
reviews
id INTEGER PK
user_id → users.id
movie_id → movies.id
rating INT 1-5
comment TEXT
created_at TEXT
